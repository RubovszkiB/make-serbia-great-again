# German mission architecture, with a one-shot outbreak and permanent vaccination guard.
MSGA_public_health = {
	MSGA_pending_outbreak = {
		icon = generic_coronavirus
		allowed = { tag = SER }
		is_good = no
		cancel_if_not_visible = yes
		days_mission_timeout = 104
		activation = { always = no }
		available = { always = no }
		visible = {
			has_country_flag = MSGA_covid_pending
			NOT = { has_country_flag = MSGA_covid_resolved }
		}
		timeout_effect = {
			clr_country_flag = MSGA_covid_pending
			if = {
				limit = {
					NOT = { has_country_flag = MSGA_covid_resolved }
					NOT = { has_country_flag = MSGA_mass_vaccination_complete }
				}
				set_country_flag = MSGA_covid_active
				if = { limit = { generic_not_has_corona = yes date < 2023.1.1 } add_ideas = low_covid_cases }
				country_event = { id = MSGA_expansion.1 }
			}
		}
	}
	MSGA_restrict_border_traffic = {
		icon = generic_propaganda_campaigns
		allowed = { tag = SER }
		visible = {
			has_country_flag = MSGA_covid_pending
			NOT = { has_country_flag = MSGA_covid_resolved }
			NOT = { has_country_flag = MSGA_restrict_border_traffic_used }
		}
		available = { has_active_mission = MSGA_pending_outbreak }
		cost = 10
		fire_only_once = yes
		ai_will_do = { factor = 10 }
		complete_effect = {
			set_country_flag = MSGA_restrict_border_traffic_used
			add_days_mission_timeout = { mission = MSGA_pending_outbreak days = 120 }
		}
	}
	MSGA_emergency_screening = {
		icon = generic_propaganda_campaigns
		allowed = { tag = SER }
		visible = {
			has_country_flag = MSGA_covid_pending
			NOT = { has_country_flag = MSGA_covid_resolved }
			NOT = { has_country_flag = MSGA_emergency_screening_used }
		}
		available = { has_active_mission = MSGA_pending_outbreak }
		cost = 15
		fire_only_once = yes
		ai_will_do = { factor = 10 }
		complete_effect = {
			set_country_flag = MSGA_emergency_screening_used
			add_days_mission_timeout = { mission = MSGA_pending_outbreak days = 130 }
		}
	}
	MSGA_public_health_controls = {
		icon = generic_propaganda_campaigns
		allowed = { tag = SER }
		visible = {
			has_country_flag = MSGA_covid_pending
			NOT = { has_country_flag = MSGA_covid_resolved }
			NOT = { has_country_flag = MSGA_public_health_controls_used }
		}
		available = { has_active_mission = MSGA_pending_outbreak }
		cost = 20
		fire_only_once = yes
		ai_will_do = { factor = 10 }
		complete_effect = {
			set_country_flag = MSGA_public_health_controls_used
			add_days_mission_timeout = { mission = MSGA_pending_outbreak days = 150 }
		}
	}
	MSGA_mass_vaccination = {
		icon = generic_propaganda_campaigns
		allowed = { tag = SER }
		visible = {
			has_country_flag = MSGA_covid_response_started
			NOT = { has_country_flag = MSGA_covid_resolved }
			NOT = { has_country_flag = MSGA_mass_vaccination_complete }
			NOT = { has_country_flag = MSGA_vaccination_started }
		}
		available = { date > 2020.12.31 }
		cost = 75
		days_remove = 20
		fire_only_once = yes
		ai_will_do = { factor = 10 }
		complete_effect = { set_country_flag = MSGA_vaccination_started }
		remove_effect = {
			MSGA_end_covid = yes
			set_country_flag = MSGA_mass_vaccination_complete
			add_stability = 0.02
			country_event = { id = MSGA_covid.3 }
		}
	}
}
