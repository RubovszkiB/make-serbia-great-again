MSGA_economic_development = {
	MSGA_invest_southern_serbia = {
		icon = generic_construction
		allowed = { tag = SER }
		visible = {
			OR = { has_completed_focus = MSGA_internal_investment_drive has_country_flag = MSGA_completed_internal_investment_drive }
			NOT = { has_country_flag = MSGA_southern_investment_started }
		}
		available = { 108 = { is_owned_by = ROOT is_fully_controlled_by = ROOT } }
		cost = 20
		days_remove = 30
		fire_only_once = yes
		complete_effect = {
			set_country_flag = MSGA_southern_investment_started
			set_temp_variable = { var = debt_var_temp value = 1.0 }
			add_debt = yes
		}
		remove_effect = {
			if = {
				limit = { 108 = { is_owned_by = ROOT is_fully_controlled_by = ROOT } }
				108 = {
					add_extra_state_shared_building_slots = 2
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
				set_country_flag = MSGA_southern_investment_complete
			}
			else = {
				# A lost state receives no factory; refund the exact nominal debt once.
				set_temp_variable = { var = debt_var_temp value = -1.0 }
				add_debt = yes
				add_political_power = 20
				custom_effect_tooltip = MSGA_southern_investment_refund_tt
			}
		}
		ai_will_do = { factor = 5 }
	}
}
