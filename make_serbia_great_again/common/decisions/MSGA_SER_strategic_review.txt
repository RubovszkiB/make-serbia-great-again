# Phase 1 briefings only. Completion flags are contracts for later Phase 2 content.
MSGA_strategic_review = {
	MSGA_assess_nato_readiness = {
		icon = generic_army_support
		allowed = {
			tag = SER
		}
		visible = {
			OR = { has_completed_focus = MSGA_watch_the_western_shield has_country_flag = MSGA_completed_watch_the_western_shield }
			NOT = {
				has_country_flag = MSGA_started_assess_nato_readiness
			}
		}
		available = {
			always = yes
		}
		cost = 25
		days_remove = 14
		ai_will_do = {
			factor = 2
		}
		complete_effect = { custom_effect_tooltip = MSGA_assess_nato_readiness_ux_complete_effect_tt hidden_effect = {
			set_country_flag = MSGA_started_assess_nato_readiness
		 } }
		remove_effect = { custom_effect_tooltip = MSGA_assess_nato_readiness_ux_remove_effect_tt hidden_effect = {
			set_country_flag = MSGA_done_assess_nato_readiness
			army_experience = 5
			add_to_variable = {
				MSGA_strategic_review_progress = 1
			}
			custom_effect_tooltip = MSGA_nato_assessment_tt
		 } }
	}
	MSGA_strengthen_border_intelligence = {
		icon = generic_army_support
		allowed = {
			tag = SER
		}
		visible = {
			OR = { has_completed_focus = MSGA_watch_the_western_shield has_country_flag = MSGA_completed_watch_the_western_shield }
			NOT = {
				has_country_flag = MSGA_started_strengthen_border_intelligence
			}
		}
		available = { custom_trigger_tooltip = { tooltip = MSGA_strengthen_border_intelligence_ux_requirements_tt hidden_trigger = {
			check_variable = {
				var = income_var
				value = MSGA_quote_border_intelligence
				compare = greater_than_or_equals
			}
		 } } }
		cost = 35
		days_remove = 30
		ai_will_do = {
			factor = 2
		}
		complete_effect = { custom_effect_tooltip = MSGA_strengthen_border_intelligence_ux_complete_effect_tt hidden_effect = {
			set_country_flag = MSGA_started_strengthen_border_intelligence
			set_temp_variable = {
				var = income_var_temp
				value = -0.03
			}
			add_income_with_inflation = yes
		 } }
		remove_effect = { custom_effect_tooltip = MSGA_strengthen_border_intelligence_ux_remove_effect_tt hidden_effect = {
			set_country_flag = MSGA_done_strengthen_border_intelligence
			add_timed_idea = {
				idea = MSGA_strategic_vigilance
				days = 180
			}
			add_to_variable = {
				MSGA_strategic_review_progress = 1
			}
			custom_effect_tooltip = MSGA_border_review_tt
		 } }
	}
}
