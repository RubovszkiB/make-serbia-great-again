
MSGA_rearmament = {
	MSGA_expand_defence = {
		icon = generic_industry
		allowed = {
			tag = SER
		}
		visible = {
			OR = { has_completed_focus = MSGA_restore_defence_production has_country_flag = MSGA_completed_restore_defence_production }

			}
		}

					include_locked = no
				}
			}
			check_variable = {
				var = income_var
				value = MSGA_quote_expand_defence
				compare = greater_than_or_equals
			}
		 } } }
		cost = 75
		days_remove = 120

			MSGA_update_quotes = yes
			set_variable = {
				var = MSGA_project_paid
				value = MSGA_quote_expand_defence
			}
			set_temp_variable = {
				var = income_var_temp
				value = -0.35
			}
			add_income_with_inflation = yes
			MSGA_begin_project = yes
		 } }
		cancel_trigger = {
			NOT = {
				1296 = {
					is_owned_by = ROOT
					is_fully_controlled_by = ROOT

						building = arms_factory
						size > 0
						include_locked = no
					}
				}
			}
		}
		cancel_effect = { custom_effect_tooltip = MSGA_expand_defence_ux_cancel_effect_tt hidden_effect = {
			MSGA_refund_project = yes
			add_political_power = 75
			MSGA_finish_project = yes
			custom_effect_tooltip = MSGA_refund_tt
		 } }
		remove_effect = { custom_effect_tooltip = MSGA_expand_defence_ux_remove_effect_tt hidden_effect = {
			if = {
				limit = {
					1296 = {
						is_owned_by = ROOT
						is_fully_controlled_by = ROOT
						free_building_slots = {
							building = arms_factory
							size > 0
							include_locked = no
						}
					}
				}
				1296 = {
					add_building_construction = {
						type = arms_factory
						level = 1
						instant_build = yes
					}
				}
				set_country_flag = MSGA_done_expand_defence
				country_event = {
					id = MSGA.10
				}
			}
			else = {
				MSGA_refund_project = yes
				add_political_power = 75
				custom_effect_tooltip = MSGA_refund_tt
			}
			MSGA_finish_project = yes
		 } }
	}
