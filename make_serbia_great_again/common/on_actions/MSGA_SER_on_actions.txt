on_actions = {
	on_startup = {
		effect = {
			SER = {
				MSGA_initialize = yes
				MSGA_resume_early_systems = yes
				MSGA_install_defence_minister = yes
				if = {
					limit = { has_completed_focus = MSGA_consolidate_the_state }
					# TFR startup can re-add the SPS modifier after this file runs.
					country_event = { id = MSGA_cleanup.1 days = 1 }
				}
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					tag = SER
					has_country_flag = MSGA_campaign_active
				}
				MSGA_update_quotes = yes
			}
		}
	}
}
