# Capital loss does not execute capitulation or settlement effects.
MSGA_prepare_pact_campaign = {
 if = { limit = { tag = SER has_country_flag = MSGA_pact_war_opened NOT = { has_country_flag = MSGA_balkan_coalition_defeated } }
  MNT = { if = { limit = { exists = yes NOT = { is_subject_of = SER } } set_country_flag = MSGA_pact_campaign_participant
   if = { limit = { is_major = no } set_major = yes set_country_flag = MSGA_pact_temporary_major }
  } }
  MAC = { if = { limit = { exists = yes NOT = { is_subject_of = SER } } set_country_flag = MSGA_pact_campaign_participant
   if = { limit = { is_major = no } set_major = yes set_country_flag = MSGA_pact_temporary_major }
  } }
  ALB = { if = { limit = { exists = yes NOT = { is_subject_of = SER } } set_country_flag = MSGA_pact_campaign_participant
   if = { limit = { is_major = no } set_major = yes set_country_flag = MSGA_pact_temporary_major }
  } }
  SLV = { if = { limit = { exists = yes NOT = { is_subject_of = SER } } set_country_flag = MSGA_pact_campaign_participant
   if = { limit = { is_major = no } set_major = yes set_country_flag = MSGA_pact_temporary_major }
  } }
  CRO = { if = { limit = { exists = yes NOT = { is_subject_of = SER } } set_country_flag = MSGA_pact_campaign_participant
   if = { limit = { is_major = no } set_major = yes set_country_flag = MSGA_pact_temporary_major }
  } }
 }
}

# ROOT = actual capitulated country; FROM = victorious country in on_capitulation.
MSGA_handle_pact_capitulation = {
 if = { limit = { NOT = { has_country_flag = MSGA_pact_real_capitulation } } set_country_flag = MSGA_pact_real_capitulation }
 if = { limit = { tag = MNT } SER = { MSGA_settle_pact_montenegro = yes } }
 if = { limit = { tag = MAC } SER = { MSGA_settle_pact_macedonia = yes } }
 if = { limit = { tag = ALB } SER = { MSGA_settle_pact_albania = yes } }
 if = { limit = { tag = SLV } SER = { MSGA_settle_pact_slovenia = yes } }
 if = { limit = { tag = CRO } SER = { MSGA_settle_pact_croatia = yes } }
}

MSGA_settle_pact_montenegro = {
 if = { limit = { tag = SER has_country_flag = MSGA_pact_war_opened NOT = { has_country_flag = MSGA_montenegro_defeated } MNT = { exists = yes has_country_flag = MSGA_pact_campaign_participant has_country_flag = MSGA_pact_real_capitulation } }
  MNT = {
   # Transfer faction leadership before leaving; the remaining members keep fighting.
   if = { limit = { is_faction_leader = yes }
    if = { limit = { MAC = { exists = yes is_in_faction_with = MNT has_capitulated = no has_war_with = SER NOT = { is_subject_of = SER } } } MAC = { set_faction_leader = yes } }
    else_if = { limit = { ALB = { exists = yes is_in_faction_with = MNT has_capitulated = no has_war_with = SER NOT = { is_subject_of = SER } } } ALB = { set_faction_leader = yes } }
    else_if = { limit = { SLV = { exists = yes is_in_faction_with = MNT has_capitulated = no has_war_with = SER NOT = { is_subject_of = SER } } } SLV = { set_faction_leader = yes } }
    else_if = { limit = { CRO = { exists = yes is_in_faction_with = MNT has_capitulated = no has_war_with = SER NOT = { is_subject_of = SER } } } CRO = { set_faction_leader = yes } }
   }
   if = { limit = { is_in_faction = yes } leave_faction = yes }
   remove_ideas = MSGA_montenegro_capital_resistance
  }
  # Documented API: ends ONLY this target's non-civil war relations.
  set_autonomy = { target = MNT autonomy_state = autonomy_puppet freedom_level = 0.5 end_wars = yes end_civil_wars = no }
  if = { limit = { MNT = { is_subject_of = SER NOT = { has_war_with = SER } } }
   MNT = {
    # Do not share Serbia's wartime faction: no automatic former-ally calls.
    if = { limit = { is_in_faction = yes } leave_faction = yes }
    set_rule = { can_decline_call_to_war = yes desc = MSGA_pact_decline_calls }
    add_ideas = MSGA_pact_postwar_neutrality add_timed_idea = { idea = MSGA_montenegro_postwar_recovery days = 90 }
    if = { limit = { has_country_flag = MSGA_pact_temporary_major } set_major = no clr_country_flag = MSGA_pact_temporary_major }
   }
   # Restore occupied HOME states, never ownership, cores or Kosovo.
   every_state = { limit = { is_owned_by = MNT controller = { OR = { tag = SER is_subject_of = SER AND = { is_in_faction_with = SER has_war_together_with = SER } } } } set_state_controller_to = MNT }
   set_country_flag = MSGA_montenegro_defeated
   country_event = { id = MSGA_pactwar.2 }
  }
 }
}
