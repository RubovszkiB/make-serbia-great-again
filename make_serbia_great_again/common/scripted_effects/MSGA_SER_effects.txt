# Uses only TFR's established treasury API; no derived GDP/debt writes.
MSGA_initialize = {
 if = {
  limit = { NOT = { has_country_flag = MSGA_bop_initialized } }
  add_power_balance_value = { id = MSGA_SER_streets_presidency value = -0.10 }
  set_country_flag = MSGA_bop_initialized
 }
 if = {
  limit = { NOT = { has_country_flag = MSGA_phase1_initialized } }
  set_variable = { MSGA_health_response = 0 }
  set_variable = { MSGA_authority_uses = 0 }
  set_variable = { MSGA_concession_uses = 0 }
  set_variable = { MSGA_order_uses = 0 }
  set_variable = { MSGA_business_bonus = 0 }
  set_variable = { MSGA_defence_output = 0 }
  set_variable = { MSGA_defence_efficiency = 0 }
  set_variable = { MSGA_defence_supply = 0 }
  set_variable = { MSGA_defence_planning = 0 }
  set_country_flag = MSGA_phase1_initialized
 }
 set_country_flag = MSGA_campaign_active
 if = {
  limit = { NOT = { has_country_flag = MSGA_elections_locked } }
  set_politics = { ruling_party = conservative elections_allowed = no }
  set_country_flag = MSGA_elections_locked
 }
 MSGA_update_quotes = yes
}
MSGA_start_recovery = {
 set_variable = { MSGA_business_bonus = 0.05 }
 add_dynamic_modifier = { modifier = MSGA_recovery_programme days = 365 }
}
MSGA_refresh_recovery = {
 clamp_variable = { var = MSGA_business_bonus min = 0.05 max = 0.10 }
 add_dynamic_modifier = { modifier = MSGA_recovery_programme days = 365 }
}
MSGA_health_progress = {
 add_to_variable = { MSGA_health_response = 1 }
 if = {
  limit = { has_country_flag = MSGA_covid_active check_variable = { MSGA_health_response > 1 } }
  remove_ideas = MSGA_pandemic_disruption
  add_ideas = MSGA_pandemic_easing
 }
 add_power_balance_value = { id = MSGA_SER_streets_presidency value = 0.02 }
}
MSGA_end_covid = {
 remove_ideas = MSGA_pandemic_disruption
 remove_ideas = MSGA_pandemic_easing
 clr_country_flag = MSGA_covid_active
 set_country_flag = MSGA_covid_resolved
}
MSGA_refresh_defence = {
 add_dynamic_modifier = { modifier = MSGA_defence_reorientation days = 365 }
}
MSGA_restore_defence = {
 if = {
  limit = { 1296 = { is_owned_by = ROOT is_controlled_by = ROOT } }
  1296 = {
   add_extra_state_shared_building_slots = 1
   add_building_construction = { type = arms_factory level = 1 instant_build = yes }
  }
 }
 else = {
  random_owned_state = {
   limit = { is_core_of = ROOT is_fully_controlled_by = ROOT }
   add_extra_state_shared_building_slots = 1
   add_building_construction = { type = arms_factory level = 1 instant_build = yes }
  }
 }
 set_variable = { MSGA_defence_output = 0.05 }
 MSGA_refresh_defence = yes
}
MSGA_supply_preparation = {
 if = {
  limit = { 108 = { is_owned_by = ROOT is_controlled_by = ROOT free_building_slots = { building = infrastructure size > 0 } } }
  108 = { add_building_construction = { type = infrastructure level = 1 instant_build = yes } }
 }
 else = { army_experience = 5 }
}
MSGA_begin_project = {
 set_country_flag = MSGA_major_project_active
 add_ideas = MSGA_capital_commitment
}
MSGA_finish_project = {
 clr_country_flag = MSGA_major_project_active
 remove_ideas = MSGA_capital_commitment
}
MSGA_check_world = {
 if = {
  limit = { MSGA_US_fracture_signal = yes NOT = { has_country_flag = MSGA_US_ORDER_COLLAPSED } }
  set_country_flag = MSGA_US_ORDER_COLLAPSED
  country_event = { id = MSGA.8 }
 }
}
