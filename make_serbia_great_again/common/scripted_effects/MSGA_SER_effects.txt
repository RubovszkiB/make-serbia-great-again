# Exact nominal debt uses TFR add_debt, not its inflation-scaled variant.
MSGA_internal_investment = {
	set_temp_variable = { var = debt_var_temp value = 2.0 }
	add_debt = yes
	custom_effect_tooltip = MSGA_investment_debt_tt
	if = {
		limit = { 45 = { is_owned_by = ROOT is_fully_controlled_by = ROOT } }
		45 = {
			# Add only missing factory slots; infrastructure respects TFR's cap of five.
			if = {
				limit = { NOT = { free_building_slots = { building = industrial_complex size > 1 } } }
				if = {
					limit = { free_building_slots = { building = industrial_complex size > 0 } }
					add_extra_state_shared_building_slots = 1
				}
				else = { add_extra_state_shared_building_slots = 2 }
			}
			add_building_construction = { type = industrial_complex level = 2 instant_build = yes }
			if = {
				limit = { free_building_slots = { building = infrastructure size > 2 } }
				add_building_construction = { type = infrastructure level = 3 instant_build = yes }
			}
			else_if = {
				limit = { free_building_slots = { building = infrastructure size > 1 } }
				add_building_construction = { type = infrastructure level = 2 instant_build = yes }
			}
			else_if = {
				limit = { free_building_slots = { building = infrastructure size > 0 } }
				add_building_construction = { type = infrastructure level = 1 instant_build = yes }
			}
		}
	}
	custom_effect_tooltip = MSGA_northern_infrastructure_tt
}

  limit = { NOT = { has_country_flag = MSGA_kosovo_war_started } country_exists = KOS KOS = { has_capitulated = no } }
  set_country_flag = MSGA_kosovo_war_started
  set_country_flag = MSGA_kosovo_war_active
  MSGA_suspend_campaign_guarantees = yes
  MSGA_detach_kosovo_faction = yes
  KOS = { add_ideas = MSGA_campaign_isolation }
  declare_war_on = { target = KOS type = annex_everything }
  785 = { add_dynamic_modifier = { modifier = MSGA_unprepared_sector } }
  1305 = { add_dynamic_modifier = { modifier = MSGA_unprepared_sector } }
  country_event = { id = MSGA_kosovo.4 days = 1 }
  country_event = { id = MSGA_kosovo.5 days = 2 }
  country_event = { id = MSGA_kosovo.6 days = 3 }
  MSGA_schedule_kosovo_observer = yes
 }
}
MSGA_schedule_kosovo_observer = {
 if = {
  limit = { has_country_flag = MSGA_kosovo_war_active NOT = { has_country_flag = MSGA_kosovo_observer_pending } }
  set_country_flag = MSGA_kosovo_observer_pending
  country_event = { id = MSGA_kosovo.99 days = 1 }
 }
}
MSGA_check_pristina = {
 if = {
  limit = {
   has_country_flag = MSGA_kosovo_war_active
   country_exists = KOS
   has_war_with = KOS
   controls_province = 14402
   NOT = { has_country_flag = MSGA_pristina_fallen }
  }
  set_country_flag = MSGA_pristina_fallen
  country_event = { id = MSGA_kosovo.7 }
  if = {
   limit = { country_exists = ALB ALB = { NOT = { has_war_with = SER } } }
   # Remove Albania from its faction BEFORE queuing the intervention event.
   MSGA_detach_albania_faction = yes
   ALB = { country_event = { id = MSGA_kosovo.20 days = 1 } }
  }
 }
}
MSGA_cleanup_kosovo_campaign = {
 clr_country_flag = MSGA_kosovo_war_active
 remove_ideas = MSGA_southern_command_mobilised
 785 = { remove_dynamic_modifier = { modifier = MSGA_unprepared_sector } }
 1305 = { remove_dynamic_modifier = { modifier = MSGA_unprepared_sector } }
 KOS = { remove_ideas = MSGA_kosovo_emergency_mobilisation remove_ideas = MSGA_campaign_isolation }
 ALB = { remove_ideas = MSGA_campaign_isolation }
 every_other_country = {
  if = {
   limit = { has_country_flag = MSGA_restore_SER_guarantee }
   diplomatic_relation = { country = SER relation = guarantee active = yes }
   clr_country_flag = MSGA_restore_SER_guarantee
  }
  if = {
   limit = { has_country_flag = MSGA_restore_ALB_guarantee }
   diplomatic_relation = { country = ALB relation = guarantee active = yes }
   clr_country_flag = MSGA_restore_ALB_guarantee
  }
 }
}
MSGA_validate_kosovo_victory = {
 if = {
  limit = {
   has_country_flag = MSGA_kosovo_resolution_done
   785 = { is_owned_by = SER is_fully_controlled_by = SER }
   1305 = { is_owned_by = SER is_fully_controlled_by = SER }
   NOT = { has_country_flag = MSGA_kosovo_victory_queued }
  }
  set_country_flag = MSGA_kosovo_victory_queued
  complete_national_focus = MSGA_kosovo_has_been_retaken
  country_event = { id = MSGA_kosovo.12 days = 3 }
 }
}
MSGA_resolve_kosovo_war = {
 if = {
  limit = { has_country_flag = MSGA_kosovo_war_active NOT = { has_country_flag = MSGA_kosovo_resolution_done } }
  set_country_flag = MSGA_kosovo_resolution_done
  annex_country = { target = KOS transfer_troops = no }
  transfer_state = 785
  transfer_state = 1305
  if = {
   limit = { has_country_flag = MSGA_albania_intervened has_war_with = ALB }
   white_peace = ALB
  }
  MSGA_cleanup_kosovo_campaign = yes
  country_event = { id = MSGA_kosovo.10 }
  if = { limit = { has_country_flag = MSGA_albania_intervened } country_event = { id = MSGA_kosovo.11 days = 1 } }
  MSGA_validate_kosovo_victory = yes
 }
}
MSGA_enter_post_kosovo_chapter = {
 if = {
  limit = {
   has_country_flag = MSGA_kosovo_victory_queued
   785 = { is_owned_by = SER }
   1305 = { is_owned_by = SER }
   NOT = { has_country_flag = MSGA_post_kosovo_active }
  }
  clr_country_flag = MSGA_kosovo_chapter_active
  set_country_flag = MSGA_post_kosovo_active
  load_focus_tree = { tree = MSGA_SER_post_kosovo keep_completed = yes }
 }
}
